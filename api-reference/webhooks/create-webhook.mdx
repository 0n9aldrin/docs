---
title: "Create Webhook"
api: "POST /api/organizations/{org_id}/projects/{project_id}/webhooks"
description: "Create a new webhook"
---

## Path Parameters

<ParamField path="org_id" type="string" required>
  Organization identifier
</ParamField>

<ParamField path="project_id" type="string" required>
  Project identifier
</ParamField>

## Request Body

<ParamField body="url" type="string" required>
  Webhook URL (must be HTTPS)
</ParamField>

<ParamField body="events" type="array" required>
  Array of event types to subscribe to
  
  **Available events:** `memory.created`, `memory.updated`, `memory.deleted`, `project.updated`
</ParamField>

<ParamField body="description" type="string">
  Optional description
</ParamField>

## Response

<ResponseField name="webhook" type="object">
  Created webhook object
</ResponseField>

<ResponseField name="testDelivery" type="object">
  Test delivery result
  
  <Expandable title="Test Delivery">
    <ResponseField name="success" type="boolean">Whether test succeeded</ResponseField>
    <ResponseField name="responseCode" type="number">HTTP response code</ResponseField>
    <ResponseField name="responseTime" type="string">Response time</ResponseField>
    <ResponseField name="timestamp" type="string">Test timestamp</ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>

```bash cURL
curl -X POST "https://api.photomem.io/api/organizations/{org_id}/projects/{project_id}/webhooks" \
  -H "Authorization: Bearer YOUR_CLERK_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "url": "https://api.example.com/webhooks/photomem",
    "events": ["memory.created", "memory.updated"],
    "description": "Production webhook"
  }'
```

```python Python
import requests

response = requests.post(
    f"https://api.photomem.io/api/organizations/{org_id}/projects/{project_id}/webhooks",
    headers={
        "Authorization": f"Bearer {clerk_token}",
        "Content-Type": "application/json"
    },
    json={
        "url": "https://api.example.com/webhooks/photomem",
        "events": ["memory.created", "memory.updated"],
        "description": "Production webhook"
    }
)

webhook = response.json()["webhook"]
print(f"Webhook created: {webhook['id']}")
print(f"Secret: {webhook['secret']}")
```

```typescript TypeScript
const response = await fetch(
  `https://api.photomem.io/api/organizations/${orgId}/projects/${projectId}/webhooks`,
  {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${clerkToken}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      url: 'https://api.example.com/webhooks/photomem',
      events: ['memory.created', 'memory.updated'],
      description: 'Production webhook'
    })
  }
);

const { webhook } = await response.json();
console.log(`Webhook created: ${webhook.id}`);
```
