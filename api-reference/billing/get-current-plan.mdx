---
title: "Get Current Plan"
api: "GET /api/organizations/{org_id}/billing/plan"
description: "Get current billing plan and usage statistics"
---

## Path Parameters

<ParamField path="org_id" type="string" required>
  The unique identifier of the organization
</ParamField>

## Response

<ResponseField name="plan" type="object">
  Current plan details
  
  <Expandable title="Plan">
    <ResponseField name="id" type="string">
      Plan identifier (e.g., "plan_free", "plan_pro")
    </ResponseField>

    <ResponseField name="name" type="string">
      Human-readable plan name
    </ResponseField>

    <ResponseField name="price" type="number">
      Monthly price in dollars
    </ResponseField>

    <ResponseField name="currency" type="string">
      Currency code (e.g., "USD")
    </ResponseField>

    <ResponseField name="interval" type="string">
      Billing interval: `month` or `year`
    </ResponseField>

    <ResponseField name="features" type="object">
      Plan features and limits
      
      <Expandable title="Features">
        <ResponseField name="apiCallsPerMonth" type="number">
          Monthly API call limit
        </ResponseField>

        <ResponseField name="storageGB" type="number">
          Storage limit in gigabytes
        </ResponseField>

        <ResponseField name="projects" type="number">
          Maximum number of projects
        </ResponseField>

        <ResponseField name="teamMembers" type="number">
          Maximum team members
        </ResponseField>

        <ResponseField name="webhooks" type="number">
          Maximum webhooks
        </ResponseField>

        <ResponseField name="support" type="string">
          Support level: `basic`, `priority`, `dedicated`
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="subscription" type="object">
  Current subscription status (null if no active subscription)
  
  <Expandable title="Subscription">
    <ResponseField name="id" type="string">
      Subscription identifier
    </ResponseField>

    <ResponseField name="status" type="string">
      Status: `active`, `canceled`, `past_due`, `trialing`
    </ResponseField>

    <ResponseField name="currentPeriodStart" type="string">
      ISO 8601 timestamp of billing period start
    </ResponseField>

    <ResponseField name="currentPeriodEnd" type="string">
      ISO 8601 timestamp of billing period end
    </ResponseField>

    <ResponseField name="cancelAtPeriodEnd" type="boolean">
      Whether subscription cancels at period end
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="usage" type="object">
  Current usage statistics
  
  <Expandable title="Usage">
    <ResponseField name="apiCalls" type="number">
      API calls used in current period
    </ResponseField>

    <ResponseField name="storageGB" type="number">
      Storage used in gigabytes
    </ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>

```bash cURL
curl -X GET "https://api.photomem.io/api/organizations/{org_id}/billing/plan" \
  -H "Authorization: Bearer YOUR_CLERK_TOKEN"
```

```python Python
import requests

response = requests.get(
    f"https://api.photomem.io/api/organizations/{org_id}/billing/plan",
    headers={"Authorization": f"Bearer {clerk_token}"}
)

data = response.json()
print(f"Plan: {data['plan']['name']}")
print(f"Price: ${data['plan']['price']}/month")
print(f"API Calls: {data['usage']['apiCalls']} / {data['plan']['features']['apiCallsPerMonth']}")
```

```typescript TypeScript
const response = await fetch(
  `https://api.photomem.io/api/organizations/${orgId}/billing/plan`,
  {
    headers: {
      'Authorization': `Bearer ${clerkToken}`
    }
  }
);

const data = await response.json();
console.log(`Plan: ${data.plan.name}`);
console.log(`Price: $${data.plan.price}/month`);
```
