---
title: "Get Analytics Dashboard"
api: "GET /api/organizations/{org_id}/analytics"
description: "Get comprehensive analytics dashboard data"
---

## Path Parameters

<ParamField path="org_id" type="string" required>
  Organization identifier
</ParamField>

## Query Parameters

<ParamField query="startDate" type="string">
  Start date in ISO 8601 format (defaults to 30 days ago)
</ParamField>

<ParamField query="endDate" type="string">
  End date in ISO 8601 format (defaults to now)
</ParamField>

<ParamField query="projectId" type="string">
  Optional project ID to filter analytics
</ParamField>

## Response

<ResponseField name="dateRange" type="object">
  <Expandable title="Date Range">
    <ResponseField name="start" type="string">Start date ISO 8601</ResponseField>
    <ResponseField name="end" type="string">End date ISO 8601</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="metrics" type="object">
  <Expandable title="Metrics">
    <ResponseField name="totalMemories" type="number">Total memories created</ResponseField>
    <ResponseField name="totalUsers" type="number">Unique users</ResponseField>
    <ResponseField name="totalApiCalls" type="number">Total API calls</ResponseField>
    <ResponseField name="averageResponseTime" type="string">Average response time</ResponseField>
    <ResponseField name="errorRate" type="number">Error rate percentage</ResponseField>
    <ResponseField name="topTags" type="array">Most used tags</ResponseField>
    <ResponseField name="topUsers" type="array">Most active users</ResponseField>
    <ResponseField name="dailyStats" type="array">Daily breakdown</ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>

```bash cURL
curl -X GET "https://api.photomem.io/api/organizations/{org_id}/analytics?startDate=2025-01-01T00:00:00Z&endDate=2025-01-31T23:59:59Z" \
  -H "Authorization: Bearer YOUR_CLERK_TOKEN"
```

```python Python
import requests
from datetime import datetime, timedelta

end_date = datetime.now()
start_date = end_date - timedelta(days=30)

response = requests.get(
    f"https://api.photomem.io/api/organizations/{org_id}/analytics",
    params={
        "startDate": start_date.isoformat(),
        "endDate": end_date.isoformat()
    },
    headers={"Authorization": f"Bearer {clerk_token}"}
)

metrics = response.json()["metrics"]
print(f"Total Memories: {metrics['totalMemories']:,}")
print(f"Top Tags: {[t['tag'] for t in metrics['topTags'][:5]]}")
```

```typescript TypeScript
const response = await fetch(
  `https://api.photomem.io/api/organizations/${orgId}/analytics`,
  {
    headers: {'Authorization': `Bearer ${clerkToken}`}
  }
);

const data = await response.json();
console.log(`Total Memories: ${data.metrics.totalMemories.toLocaleString()}`);
```
