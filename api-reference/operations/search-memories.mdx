---
title: "Search Memories"
api: "POST /v1/search"
description: "Search memories semantically using natural language"
---

## Request Body

<ParamField body="query" type="string" required>
  Natural language search query
</ParamField>

<ParamField body="user_id" type="string">
  Filter by user identifier
</ParamField>

<ParamField body="agent_id" type="string">
  Filter by agent identifier
</ParamField>

<ParamField body="organization_id" type="string">
  Filter by organization
</ParamField>

<ParamField body="project_id" type="string">
  Filter by project
</ParamField>

<ParamField body="limit" type="integer" default={10}>
  Maximum number of results (1-100)
</ParamField>

<ParamField body="threshold" type="float" default={0.7}>
  Minimum similarity score (0.0-1.0)
</ParamField>

## Response

<ResponseField name="results" type="array">
  Array of matching memories, ordered by relevance
  
  <Expandable title="Search Result">
    <ResponseField name="id" type="string">
      Memory identifier
    </ResponseField>
    <ResponseField name="memory" type="string">
      Memory content
    </ResponseField>
    <ResponseField name="score" type="float">
      Similarity score (0.0-1.0)
    </ResponseField>
    <ResponseField name="metadata" type="object">
      Memory metadata
    </ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>

```bash cURL
curl -X POST https://api.photomem.ai/v1/search \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "query": "outdoor activities",
    "user_id": "phone:15103651277",
    "limit": 5
  }'
```

```python Python
results = await memory.search(
    query="outdoor activities",
    user_id="phone:15103651277",
    limit=5
)
```

</RequestExample>
