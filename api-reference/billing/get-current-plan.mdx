---
title: 'Get Current Plan'
openapi: get /api/organizations/{org_id}/billing/plan
---

Get the current subscription plan, features, and usage information for an organization.

<Info>
View your plan details, feature limits, and current usage to understand your PhotoMem subscription.
</Info>

## Authentication

This endpoint requires JWT authentication.

```bash
Authorization: Bearer YOUR_JWT_TOKEN
```

## Path Parameters

<ParamField path="org_id" type="string" required>
  The unique identifier of the organization
</ParamField>

## Response

<ResponseField name="plan" type="object">
  <Expandable title="Plan Details">
    <ResponseField name="id" type="string">
      Plan identifier (e.g., "plan_free", "plan_pro")
    </ResponseField>
    
    <ResponseField name="name" type="string">
      Human-readable plan name
    </ResponseField>
    
    <ResponseField name="price" type="number">
      Monthly price in USD
    </ResponseField>
    
    <ResponseField name="currency" type="string">
      Currency code (USD)
    </ResponseField>
    
    <ResponseField name="interval" type="string">
      Billing interval ("month" or "year")
    </ResponseField>
    
    <ResponseField name="features" type="object">
      Plan feature limits and capabilities
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="subscription" type="object">
  <Expandable title="Subscription Status">
    <ResponseField name="id" type="string">
      Subscription identifier
    </ResponseField>
    
    <ResponseField name="status" type="string">
      Subscription status: `active`, `trialing`, `canceled`, `past_due`
    </ResponseField>
    
    <ResponseField name="currentPeriodStart" type="string">
      ISO 8601 timestamp of current billing period start
    </ResponseField>
    
    <ResponseField name="currentPeriodEnd" type="string">
      ISO 8601 timestamp of current billing period end
    </ResponseField>
    
    <ResponseField name="cancelAtPeriodEnd" type="boolean">
      Whether subscription will cancel at end of current period
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="usage" type="object">
  Current usage statistics for the billing period
</ResponseField>

## Subscription Status Values

<AccordionGroup>
  <Accordion title="active">
    Subscription is active and in good standing. All features are available.
  </Accordion>
  
  <Accordion title="trialing">
    In trial period. All paid features are available until trial ends.
  </Accordion>
  
  <Accordion title="canceled">
    Subscription has been canceled. Access continues until `currentPeriodEnd`.
  </Accordion>
  
  <Accordion title="past_due">
    Payment failed. Features may be limited. Update payment method to restore access.
  </Accordion>
</AccordionGroup>

## Usage Monitoring

Track your usage against plan limits:

```typescript
async function checkPlanLimits(orgId: string, token: string) {
  const { plan, usage } = await fetch(
    `/api/organizations/${orgId}/billing/plan`,
    { headers: { 'Authorization': `Bearer ${token}` } }
  ).then(r => r.json());

  const memoryUsagePercent = (usage.memoriesCreated / plan.features.memoriesPerMonth) * 100;
  const searchUsagePercent = (usage.searchesPerformed / plan.features.searchesPerMonth) * 100;
  const storageUsagePercent = (usage.storageUsedGB / plan.features.storageGB) * 100;

  return {
    memories: {
      used: usage.memoriesCreated,
      limit: plan.features.memoriesPerMonth,
      percent: memoryUsagePercent.toFixed(1)
    },
    searches: {
      used: usage.searchesPerformed,
      limit: plan.features.searchesPerMonth,
      percent: searchUsagePercent.toFixed(1)
    },
    storage: {
      used: usage.storageUsedGB,
      limit: plan.features.storageGB,
      percent: storageUsagePercent.toFixed(1)
    }
  };
}
```

## Upgrade Prompts

Show upgrade prompts based on usage:

```typescript
function shouldShowUpgradePrompt(planData: any): boolean {
  const { plan, usage } = planData;
  
  // Show if using > 80% of any limit
  const memoryUsage = usage.memoriesCreated / plan.features.memoriesPerMonth;
  const searchUsage = usage.searchesPerformed / plan.features.searchesPerMonth;
  const storageUsage = usage.storageUsedGB / plan.features.storageGB;
  
  return memoryUsage > 0.8 || searchUsage > 0.8 || storageUsage > 0.8;
}
```
