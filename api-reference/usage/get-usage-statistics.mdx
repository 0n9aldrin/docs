---
title: 'Get Usage Statistics'
openapi: get /api/organizations/{org_id}/usage
---

Get comprehensive usage statistics for an organization, including API calls, storage metrics, and cost estimates.

<Info>
Track your PhotoMem usage across all projects to optimize costs and monitor API consumption patterns.
</Info>

## Authentication

This endpoint requires JWT authentication.

```bash
Authorization: Bearer YOUR_JWT_TOKEN
```

## Path Parameters

<ParamField path="org_id" type="string" required>
  The unique identifier of the organization
</ParamField>

## Query Parameters

<ParamField query="period" type="string" default="30d">
  Time period for statistics. Options: `7d`, `30d`, `90d`, `current_month`
</ParamField>

<ParamField query="projectId" type="string">
  Optional project ID to filter statistics to a specific project
</ParamField>

## Response

<ResponseField name="period" type="object">
  <Expandable title="Period Information">
    <ResponseField name="start" type="string">
      ISO 8601 timestamp of period start
    </ResponseField>
    
    <ResponseField name="end" type="string">
      ISO 8601 timestamp of period end
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="usage" type="object">
  <Expandable title="Usage Metrics">
    <ResponseField name="memoriesCreated" type="integer">
      Total number of memories created in this period
    </ResponseField>
    
    <ResponseField name="memoriesSearched" type="integer">
      Total number of search operations performed
    </ResponseField>
    
    <ResponseField name="storageBytes" type="integer">
      Total storage used in bytes
    </ResponseField>
    
    <ResponseField name="estimatedCost" type="number">
      Estimated cost in USD for the period
    </ResponseField>
  </Expandable>
</ResponseField>

## Usage Dashboard Example

Build a usage dashboard with this data:

```typescript
async function buildUsageDashboard(orgId: string, jwtToken: string) {
  // Get different time periods
  const [week, month, quarter] = await Promise.all([
    fetch(`/api/organizations/${orgId}/usage?period=7d`, {
      headers: { 'Authorization': `Bearer ${jwtToken}` }
    }).then(r => r.json()),
    fetch(`/api/organizations/${orgId}/usage?period=30d`, {
      headers: { 'Authorization': `Bearer ${jwtToken}` }
    }).then(r => r.json()),
    fetch(`/api/organizations/${orgId}/usage?period=90d`, {
      headers: { 'Authorization': `Bearer ${jwtToken}` }
    }).then(r => r.json())
  ]);

  return {
    weekly: {
      memories: week.usage.memoriesCreated,
      searches: week.usage.memoriesSearched,
      cost: week.usage.estimatedCost
    },
    monthly: {
      memories: month.usage.memoriesCreated,
      searches: month.usage.memoriesSearched,
      cost: month.usage.estimatedCost
    },
    quarterly: {
      memories: quarter.usage.memoriesCreated,
      searches: quarter.usage.memoriesSearched,
      cost: quarter.usage.estimatedCost
    },
    trends: {
      memoriesPerDay: month.usage.memoriesCreated / 30,
      searchesPerDay: month.usage.memoriesSearched / 30,
      avgCostPerMemory: month.usage.estimatedCost / month.usage.memoriesCreated
    }
  };
}
```

## Monitoring & Alerts

Use this endpoint to implement usage monitoring:

```typescript
async function checkUsageAlerts(orgId: string, token: string) {
  const usage = await fetch(
    `/api/organizations/${orgId}/usage?period=current_month`,
    { headers: { 'Authorization': `Bearer ${token}` } }
  ).then(r => r.json());

  const alerts = [];

  // Check if approaching plan limits
  if (usage.usage.memoriesCreated > 9000) { // 90% of 10k limit
    alerts.push({
      level: 'warning',
      message: 'Approaching monthly memory limit'
    });
  }

  // Check for unusual cost spike
  if (usage.usage.estimatedCost > 100) {
    alerts.push({
      level: 'warning',
      message: 'Unusually high cost this month'
    });
  }

  return alerts;
}
```

## Storage Calculation

The `storageBytes` value represents the total size of:
- Memory content (raw text)
- Vector embeddings
- Metadata and tags
- Change history (if enabled)

To convert to human-readable format:

```typescript
function formatBytes(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(2)} KB`;
  if (bytes < 1024 * 1024 * 1024) return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
  return `${(bytes / (1024 * 1024 * 1024)).toFixed(2)} GB`;
}

console.log(formatBytes(data.usage.storageBytes)); // "50.00 MB"
```
