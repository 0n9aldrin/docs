---
title: "Change Plan"
api: "POST /api/organizations/{org_id}/billing/plan/change"
description: "Change billing plan with automatic proration"
---

## Path Parameters

<ParamField path="org_id" type="string" required>
  The unique identifier of the organization
</ParamField>

## Request Body

<ParamField body="planId" type="string" required>
  The plan identifier to change to
  
  **Available options:** `plan_free`, `plan_starter`, `plan_pro`, `plan_enterprise`
</ParamField>

<ParamField body="interval" type="string" default="month">
  Billing interval
  
  **Available options:** `month`, `year`
</ParamField>

## Response

<ResponseField name="subscription" type="object">
  Updated subscription details
  
  <Expandable title="Subscription">
    <ResponseField name="id" type="string">
      Subscription identifier
    </ResponseField>

    <ResponseField name="planId" type="string">
      New plan identifier
    </ResponseField>

    <ResponseField name="status" type="string">
      Subscription status
    </ResponseField>

    <ResponseField name="nextBillingDate" type="string">
      ISO 8601 timestamp of next billing date
    </ResponseField>

    <ResponseField name="prorationAmount" type="number">
      Amount charged or credited due to plan change
    </ResponseField>

    <ResponseField name="currency" type="string">
      Currency code
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="invoice" type="object">
  Invoice for proration charge (null if no charge)
  
  <Expandable title="Invoice">
    <ResponseField name="id" type="string">
      Invoice identifier
    </ResponseField>

    <ResponseField name="amount" type="number">
      Invoice amount
    </ResponseField>

    <ResponseField name="dueDate" type="string">
      ISO 8601 timestamp
    </ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>

```bash cURL
curl -X POST "https://api.photomem.io/api/organizations/{org_id}/billing/plan/change" \
  -H "Authorization: Bearer YOUR_CLERK_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "planId": "plan_pro",
    "interval": "month"
  }'
```

```python Python
import requests

response = requests.post(
    f"https://api.photomem.io/api/organizations/{org_id}/billing/plan/change",
    headers={
        "Authorization": f"Bearer {clerk_token}",
        "Content-Type": "application/json"
    },
    json={
        "planId": "plan_pro",
        "interval": "month"
    }
)

result = response.json()
print(f"Plan changed to: {result['subscription']['planId']}")
print(f"Proration: ${result['subscription']['prorationAmount']}")
```

```typescript TypeScript
const response = await fetch(
  `https://api.photomem.io/api/organizations/${orgId}/billing/plan/change`,
  {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${clerkToken}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      planId: 'plan_pro',
      interval: 'month'
    })
  }
);

const result = await response.json();
console.log(`Plan changed: ${result.subscription.planId}`);
```
