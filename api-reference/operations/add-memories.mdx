---
title: "Add Memories"
api: "POST /v1/memories"
description: "Create new memories from messages"
---

## Request Body

<ParamField body="messages" type="array" required>
  Array of messages to extract memories from
  
  <Expandable title="Message Object">
    <ParamField body="role" type="string" required>
      Message role: `user`, `assistant`, or `system`
    </ParamField>
    <ParamField body="content" type="string" required>
      Message content
    </ParamField>
  </Expandable>
</ParamField>

<ParamField body="organization_id" type="string">
  Organization identifier for multi-tenancy
</ParamField>

<ParamField body="project_id" type="string">
  Project identifier for organizing memories
</ParamField>

<ParamField body="user_id" type="string">
  User identifier (format: `type:value` e.g., `phone:123`)
</ParamField>

<ParamField body="agent_id" type="string">
  Agent identifier
</ParamField>

<ParamField body="app_id" type="string">
  Application identifier
</ParamField>

<ParamField body="run_id" type="string">
  Run/session identifier
</ParamField>

<ParamField body="entity_type" type="enum<string>">
  Type of entity creating the memory
  
  **Available options:** `user`, `agent`, `app`, `run`
</ParamField>

<ParamField body="metadata" type="object">
  Additional metadata to store with the memory
</ParamField>

<ParamField body="tags" type="array">
  Tags for categorizing memories
</ParamField>

<ParamField body="memory_type" type="string" default="semantic">
  Type of memory storage
</ParamField>

<ParamField body="infer_entities" type="boolean" default={true}>
  Extract entities from the messages
</ParamField>

<ParamField body="build_graph" type="boolean" default={true}>
  Build knowledge graph relationships
</ParamField>

<ParamField body="deduplicate" type="boolean" default={true}>
  Check for and skip duplicate memories
</ParamField>

## Response

<ResponseField name="results" type="array">
  Array of created memory objects
  
  <Expandable title="Memory Object">
    <ResponseField name="id" type="string">
      Unique memory identifier
    </ResponseField>
    <ResponseField name="memory" type="string">
      The extracted memory text
    </ResponseField>
    <ResponseField name="event" type="string">
      Event type: `ADD`, `UPDATE`, or `DELETE`
    </ResponseField>
    <ResponseField name="hash" type="string">
      Content hash for deduplication
    </ResponseField>
    <ResponseField name="metadata" type="object">
      Memory metadata including timestamps, tags, and identifiers
    </ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>

```bash cURL
curl -X POST https://api.photomem.ai/v1/memories \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "messages": [
      {"role": "user", "content": "I love hiking in the Swiss Alps"}
    ],
    "user_id": "phone:15103651277",
    "entity_type": "user",
    "tags": ["hobbies", "outdoor"]
  }'
```

```python Python
from photomem import Memory

memory = Memory(api_key="YOUR_API_KEY")

result = await memory.add(
    messages=[
        {"role": "user", "content": "I love hiking in the Swiss Alps"}
    ],
    user_id="phone:15103651277",
    entity_type="user",
    tags=["hobbies", "outdoor"]
)
```

```typescript TypeScript
import { Memory } from 'photomem';

const memory = new Memory({ apiKey: 'YOUR_API_KEY' });

const result = await memory.add({
  messages: [
    { role: 'user', content: 'I love hiking in the Swiss Alps' }
  ],
  userId: 'phone:15103651277',
  entityType: 'user',
  tags: ['hobbies', 'outdoor']
});
```
