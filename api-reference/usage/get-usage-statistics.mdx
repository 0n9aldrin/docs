---
title: "Get Usage Statistics"
api: "GET /api/organizations/{org_id}/usage"
description: "Get detailed usage statistics for an organization"
---

## Path Parameters

<ParamField path="org_id" type="string" required>
  Organization identifier
</ParamField>

## Query Parameters

<ParamField query="period" type="string" default="30d">
  Time period
  
  **Available options:** `7d`, `30d`, `90d`, `current_month`
</ParamField>

<ParamField query="projectId" type="string">
  Optional project ID to filter statistics
</ParamField>

## Response

<ResponseField name="period" type="string">
  The time period for the statistics
</ResponseField>

<ResponseField name="usage" type="object">
  Usage metrics
  
  <Expandable title="Usage">
    <ResponseField name="apiCalls" type="object">
      API call statistics
      
      <Expandable title="API Calls">
        <ResponseField name="total" type="number">Total API calls</ResponseField>
        <ResponseField name="successful" type="number">Successful calls</ResponseField>
        <ResponseField name="failed" type="number">Failed calls</ResponseField>
        <ResponseField name="breakdown" type="object">Breakdown by operation type</ResponseField>
      </Expandable>
    </ResponseField>
    
    <ResponseField name="storage" type="object">
      Storage usage metrics
      
      <Expandable title="Storage">
        <ResponseField name="totalBytes" type="number">Total storage in bytes</ResponseField>
        <ResponseField name="memoryCount" type="number">Number of memories</ResponseField>
        <ResponseField name="averageMemorySize" type="number">Average memory size in bytes</ResponseField>
      </Expandable>
    </ResponseField>
    
    <ResponseField name="costs" type="object">
      Cost breakdown
      
      <Expandable title="Costs">
        <ResponseField name="apiCalls" type="number">API call costs</ResponseField>
        <ResponseField name="storage" type="number">Storage costs</ResponseField>
        <ResponseField name="total" type="number">Total costs</ResponseField>
        <ResponseField name="currency" type="string">Currency code</ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="limits" type="object">
  Plan limits and usage percentages
  
  <Expandable title="Limits">
    <ResponseField name="apiCallsPerMonth" type="number">Monthly API call limit</ResponseField>
    <ResponseField name="storageGB" type="number">Storage limit in GB</ResponseField>
    <ResponseField name="percentUsed" type="object">Usage percentages</ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>

```bash cURL
curl -X GET "https://api.photomem.io/api/organizations/{org_id}/usage?period=30d" \
  -H "Authorization: Bearer YOUR_CLERK_TOKEN"
```

```python Python
import requests

response = requests.get(
    f"https://api.photomem.io/api/organizations/{org_id}/usage",
    params={"period": "30d"},
    headers={"Authorization": f"Bearer {clerk_token}"}
)

data = response.json()
print(f"API Calls: {data['usage']['apiCalls']['total']:,}")
print(f"Total Cost: ${data['usage']['costs']['total']:.2f}")
```

```typescript TypeScript
const response = await fetch(
  `https://api.photomem.io/api/organizations/${orgId}/usage?period=30d`,
  {
    headers: {'Authorization': `Bearer ${clerkToken}`}
  }
);

const data = await response.json();
console.log(`API Calls: ${data.usage.apiCalls.total.toLocaleString()}`);
```
