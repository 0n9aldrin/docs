---
title: "Get Billing History"
api: "GET /api/organizations/{org_id}/billing/invoices"
description: "Get organization's billing history and invoices"
---

## Path Parameters

<ParamField path="org_id" type="string" required>
  The unique identifier of the organization
</ParamField>

## Response

<ResponseField name="invoices" type="array">
  Array of invoice objects
  
  <Expandable title="Invoice">
    <ResponseField name="id" type="string">
      Unique invoice identifier
    </ResponseField>

    <ResponseField name="number" type="string">
      Human-readable invoice number (e.g., "INV-2025-001")
    </ResponseField>

    <ResponseField name="amount" type="number">
      Invoice amount in dollars
    </ResponseField>

    <ResponseField name="currency" type="string">
      Currency code (e.g., "USD")
    </ResponseField>

    <ResponseField name="status" type="string">
      Invoice status: `paid`, `pending`, `failed`, `refunded`
    </ResponseField>

    <ResponseField name="dueDate" type="string">
      ISO 8601 timestamp of due date
    </ResponseField>

    <ResponseField name="paidAt" type="string">
      ISO 8601 timestamp of payment (null if unpaid)
    </ResponseField>

    <ResponseField name="downloadUrl" type="string">
      URL to download PDF invoice
    </ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>

```bash cURL
curl -X GET "https://api.photomem.io/api/organizations/{org_id}/billing/invoices" \
  -H "Authorization: Bearer YOUR_CLERK_TOKEN"
```

```python Python
import requests

response = requests.get(
    f"https://api.photomem.io/api/organizations/{org_id}/billing/invoices",
    headers={"Authorization": f"Bearer {clerk_token}"}
)

invoices = response.json()["invoices"]
for invoice in invoices:
    print(f"{invoice['number']}: ${invoice['amount']} - {invoice['status']}")
```

```typescript TypeScript
const response = await fetch(
  `https://api.photomem.io/api/organizations/${orgId}/billing/invoices`,
  {
    headers: {
      'Authorization': `Bearer ${clerkToken}`
    }
  }
);

const { invoices } = await response.json();
invoices.forEach(inv => {
  console.log(`${inv.number}: $${inv.amount} - ${inv.status}`);
});
```
